  'Current:',
  'Expected',
  'Current',

bool _isClean(String s) {
  final lower = s.toLowerCase();
  for (final token in _forbiddenTokens) {
    if (lower.contains(token.toLowerCase())) {
      return false;
    }
  }
  return true;
}

String _emptyClean(String text) {
  return _isClean(text) ? text : 'Requirement';
}



String _explainPassed(EligibilityRuleResult rule) {
  final friendly = _asFriendlyLabel(rule);
  if (rule.looksDocumentRelated) {
    return 'We have accepted your $friendly for this scheme.';
  }
  return 'Your ${friendly.toLowerCase()} matches this scheme\'s requirement.';
}

String _explainMissingProfile(EligibilityRuleResult rule) {
  final friendly = _asFriendlyLabel(rule);
  if (friendly.isEmpty) {
    return 'We need some information from your profile before we can confirm your eligibility.';
  }
  return 'We need to know $friendly before we can confirm your eligibility.';
}

String _explainMissingDocument(EligibilityRuleResult rule) {
  final friendly = _asFriendlyLabel(rule);
  if (friendly.isEmpty) {
    return 'You may need to provide a supporting document for this scheme.';
  }
  return 'You still need to provide $friendly for this scheme.';
}



String _asFriendlyLabel(EligibilityRuleResult rule) {
  final desc = rule.description?.trim();
  if (desc.isNotEmpty && !desc.contains('Expected') && !desc.contains('Current')) {
    return _emptyClean(_cleanupLabel(desc));
  }

  final condition = rule.condition.trim();
  if (condition.isNotEmpty) {
    return _emptyClean(_conditionToNounPhrase(condition));
  }

  final code = rule.ruleCode.trim();
  if (code.isNotEmpty) {
    return _emptyClean(_humanize(code));
  }

  return 'requirement';
}

String _cleanupLabel(String text) {
  var s = text.trim();
  s = s.replaceAll(RegExp(r'\s+'), ' ');
  return s;
}

String _humanize(String text) {
  var s = text.trim();
  s = s.replaceAll(RegExp(r'[-_./]+'), ' ');
  s = s.toLowerCase();
  if (s.startsWith('the ')) {
    s = s.substring(4);
  }
  s = s.replaceFirstMapped(
      RegExp(r'\b(age|income|status|level|amount)\b'), (m) => m.group(0)!.toUpperCase());
  return _capitalize(s.trim());
}

String _capitalize(String text) {
  if (text.isEmpty) return text;
  return text[0].toUpperCase() + text.substring(1);
}

String _conditionToNounPhrase(String condition) {
  var s = condition.trim();
  s = s.replaceAll(RegExp(r'\s+'), ' ');
  s = s.replaceAllMapped(
      RegExp(r'\b(?:>=?|<=?|==|!=|not equal to|equal to|greater than|less than)\b'),
      (m) => ' ');
  final parts = s.split(RegExp(r'[\s,;]+')).where((p) => p.isNotEmpty).toList();
  final nouns = <String>[];
  for (final p in parts) {
    final cleaned = _capitalize(p.trim().toLowerCase());
    if (cleaned.isNotEmpty &&
        cleaned.length < 60 &&
        !_forbiddenTokens.any((t) => cleaned.toLowerCase().contains(t.toLowerCase()))) {
      nouns.add(cleaned);
    }
  }
  if (nouns.isEmpty) return 'requirement';
  return nouns.take(3).join(' ').trim();
}

class EligibilityPresentationGuard {
  const EligibilityPresentationGuard._();

  static bool isCitizenSafe(Object? value) {
    if (value is! String) return true;
    final v = value.trim();
    if (v.isEmpty) return true;
    final lower = v.toLowerCase();
    for (final token in _forbiddenTokens) {
      if (lower.contains(token.toLowerCase())) {
        return false;
      }
    }
    return true;
  }

  static List<String> safeLines(List<String> lines) {
    return lines
        .where((line) => _isClean(line.trim()))
        .map((line) => line.trim())
        .where((line) => line.isNotEmpty)
        .toList();
  }
}



String _makeBenefitSummary(String reasoning) {
  final trimmed = reasoning.trim();
  if (trimmed.isEmpty) {
    return 'This scheme may provide benefits relevant to you.';
  }
  if (trimmed.length > 160) {
    return trimmed.substring(0, 157) + '...';
  }
  return trimmed;
}

String _nextAction(String status, bool applicationReady,
    bool hasMissingInfo, bool hasMissingDocs) {
  if (status == 'manual_review_required') {
    return 'Your application may need manual verification. Please keep an eye on updates.';
  }
  if (status == 'not_eligible') {
    return 'Review the information above and complete any missing details where possible.';
  }
  if (hasMissingDocs) {
    return 'Upload the documents listed above to continue.';
  }
  if (hasMissingInfo) {
    return 'Update your profile so we can confirm your eligibility.';
  }
  if (applicationReady) {
    return 'Your application looks ready.';
  }
  return 'Complete your profile and documents to continue.';
}

String _normalizeStatus(String? status, bool eligible) {
  final s = (status ?? '').toLowerCase().trim();
  if (s.isEmpty) return eligible ? 'eligible' : 'not_eligible';
  if (s.startsWith('eligible')) return 'eligible';
  if (s == 'not_eligible' || s == 'not eligible') return 'not_eligible';
  if (s == 'manual_review_required' ||
      s == 'manual review required' ||
      s == 'manual_review' ||
      s == 'manual') {
    return 'manual_review_required';
  }
  if (s == 'insufficient_information' ||
      s == 'insufficient information' ||
      s == 'missing' ||
      s == 'partial') {
    return 'insufficient_information';
  }
  return s;
}

String _statusLabel(String status, bool applicationReady) {
  switch (status) {
    case 'eligible':
      return applicationReady
          ? 'Your information matches this scheme'
          : 'You may be eligible';
    case 'insufficient_information':
      return 'More information needed';
    case 'not_eligible':
      return 'You are not currently eligible';
    case 'manual_review_required':
      return 'Manual verification required';
    default:
      return 'Information not available';
  }
}

// Presenter layer: translates structured eligibility results into citizen-friendly
// explanations. This layer MUST NOT make eligibility decisions; it only restates
// decisions already made by the EligibilityEngine / EligibilityEvaluator.
//
// Raw internal fields (rule_code, field names, expected_value / actual_value dumps,
// source chunk text, retrieval metadata) are intentionally NOT surfaced through this
// presenter into the normal citizen UI.
library eligibility_presenter;

import 'package:flutter/foundation.dart';

import '../../models/eligibility.dart';

@immutable
class EligibilityPresentation {
  const EligibilityPresentation({
    required this.status,
    required this.statusLabel,
    required this.whyYouMatch,
    required this.missingInformation,
    this.missingDocuments = const [],
    required this.benefitSummary,
    required this.nextAction,
    this.rawEligibility,
  });

  final String status;
  final String statusLabel;
  final List<String> whyYouMatch;
  final List<String> missingInformation;
  final List<String> missingDocuments;
  final String benefitSummary;
  final String nextAction;

  /// Raw structured eligibility for a possible future detailed view/debug mode.
  /// Not rendered in the normal citizen card.
  final EligibilityCheck? rawEligibility;
}

EligibilityPresentation presentEligibility(EligibilityCheck check) {
  final status = _normalizeStatus(check.eligibilityStatus, check.eligible);
  final statusLabel = _statusLabel(status, check.applicationReady);

  final whyYouMatch = <String>[];
  final missingInformation = <String>[];
  final missingDocuments = <String>[];

  final passed = check.matchedRules.where((r) => r.passed).toList();
  final failed = check.failedRules.where((r) => r.passed == false).toList();

  for (final rule in passed.take(6)) {
    final sentence = _explainPassed(rule);
    if (sentence.isNotEmpty) whyYouMatch.add(sentence);
  }

  for (final rule in failed) {
    if (rule.hasMissingProfileValue) {
      final sentence = _explainMissingProfile(rule);
      if (sentence.isNotEmpty) missingInformation.add(sentence);
    } else {
      final sentence = _explainMissingDocument(rule);
      if (sentence.isNotEmpty) missingDocuments.add(sentence);
    }
  }

  final benefitSummary = check.reasoning.isNotEmpty
      ? _makeBenefitSummary(check.reasoning)
      : 'This scheme may provide benefits relevant to you.';

  final nextAction = _nextAction(status, check.applicationReady,
      missingInformation.isNotEmpty, missingDocuments.isNotEmpty);

  assert(whyYouMatch.every(_isClean),
      'Citizen sentences leaked internal tokens: ${whyYouMatch.where((s) => !_isClean(s))}');
  assert(missingInformation.every(_isClean),
      'Citizen sentences leaked internal tokens: ${missingInformation.where((s) => !_isClean(s))}');
  assert(missingDocuments.every(_isClean),
      'Citizen sentences leaked internal tokens: ${missingDocuments.where((s) => !_isClean(s))}');

  return EligibilityPresentation(
    status: status,
    statusLabel: statusLabel,
    whyYouMatch: whyYouMatch,
    missingInformation: missingInformation,
    missingDocuments: missingDocuments,
    benefitSummary: benefitSummary,
    nextAction: nextAction,
    rawEligibility: check,
  );
}

/// Citizen-facing eligibility presentation layer.
///
/// This mapper is the ONLY place where structured eligibility results
/// (fields, operators, expected/actual values, PASS/FAIL/UNKNOWN, source
/// documents) are translated into human-readable explanations.
///
/// Hard rules:
/// - It NEVER makes eligibility decisions. It only explains outcomes the
///   backend [EligibilityEvaluator] already decided (`passed` booleans and
///   the engine's `eligibility_status` are consumed as-is).
/// - It NEVER leaks raw internals into citizen text: no snake_case field
///   names, no `Expected: [...]` / `Current: [...]` dumps, no operator
///   symbols, no rule ids, no source chunk text.
/// - Detailed technical information is preserved in [DetailedRuleInfo] so a
///   review/debug surface can show it; citizen UI renders only
///   [EligibilityPresentation] citizen fields.
library;
import '../utils/evidence_mapping.dart';

/// Visual tone a screen should use when rendering a line/status.
enum PresentationTone { positive, caution, negative, neutral }

/// One citizen-readable explanation sentence.
class ExplanationLine {
  const ExplanationLine(this.text, this.tone);
  final String text;
  final PresentationTone tone;
}

/// Technical detail preserved for review/debug surfaces. Never rendered in
/// the normal citizen UI.
class DetailedRuleInfo {
  const DetailedRuleInfo({
    required this.field,
    required this.operatorName,
    required this.expected,
    required this.actual,
    required this.resultLabel,
    this.sourceDocument,
  });
  final String field;
  final String operatorName;
  final String expected;
  final String actual;
  final String resultLabel;
  final String? sourceDocument;
}

/// Simple presentation model for one scheme's eligibility decision.
class EligibilityPresentation {
  const EligibilityPresentation({
    required this.status,
    required this.statusLabel,
    required this.statusTone,
    required this.whyYouMatch,
    required this.missingInformation,
    required this.missingDocuments,
    this.benefitSummary,
    this.nextAction,
    this.detailedInformation = const [],
  });

  /// Raw engine status, kept verbatim for debug/telemetry only.
  final String status;
  final String statusLabel;
  final PresentationTone statusTone;
  final List<ExplanationLine> whyYouMatch;
  final List<ExplanationLine> missingInformation;
  final List<ExplanationLine> missingDocuments;
  final String? benefitSummary;
  final String? nextAction;

  /// Optional technical mirror (rule ids, operators, expected/actual,
  /// source documents) for a detailed/debug page.
  final List<DetailedRuleInfo> detailedInformation;
}

const Map<String, String> _statusLabels = {
  'eligible': 'You are eligible',
  'potentially_eligible': 'You may be eligible',
  'possibly_eligible': 'You may be eligible',
  'insufficient_information': 'More information needed',
  'not_eligible': 'You are not currently eligible',
  'ineligible': 'You are not currently eligible',
  'manual_review_required': 'Manual verification required',
};

/// Engine status → citizen status label. Unknown statuses fall back to a
/// readable title-case version; they are never shown raw.
String statusLabelFor(String status) {
  final key = status.trim().toLowerCase().replaceAll(' ', '_');
  final label = _statusLabels[key];
  if (label != null) return label;
  final text = key.replaceAll('_', ' ').trim();
  if (text.isEmpty) return 'Eligibility status not available';
  return text
      .split(' ')
      .map((w) => w.isEmpty ? w : '${w[0].toUpperCase()}${w.substring(1)}')
      .join(' ');
}

PresentationTone statusToneFor(String status) {
  switch (status.trim().toLowerCase().replaceAll(' ', '_')) {
    case 'eligible':
      return PresentationTone.positive;
    case 'potentially_eligible':
    case 'possibly_eligible':
    case 'insufficient_information':
    case 'manual_review_required':
      return PresentationTone.caution;
    case 'not_eligible':
    case 'ineligible':
      return PresentationTone.negative;
    default:
      return PresentationTone.neutral;
  }
}

/// Builds a citizen-facing eligibility presentation from structured results.
///
/// [rules] is the structured condition-level evaluation the backend already
/// decided. [status] is the engine-level summary status.
///
/// This function does not recompute eligibility. It only explains the backend
/// outcomes in citizen language.
EligibilityPresentation buildEligibilityPresentation({
  required List<EligibilityRuleResult> rules,
  required String status,
  String? benefitSummary,
}) {
  final list = <ExplanationLine>[];
  final details = <DetailedRuleInfo>[];
  final missingDocuments = <ExplanationLine>[];

  for (final rule in rules) {
    final line = _buildLine(rule);
    list.add(line);
    details.add(_buildDetail(rule));

    if ((line.tone == PresentationTone.negative ||
            line.tone == PresentationTone.neutral) &&
        _likelyDocument(rule.fieldName)) {
      missingDocuments.add(line);
    }
  }

  final missingInformation = list.where((l) {
    return (l.tone == PresentationTone.negative ||
            l.tone == PresentationTone.neutral) &&
        !missingDocuments.contains(l);
  }).toList();

  return EligibilityPresentation(
    status: status.trim(),
    statusLabel: statusLabelFor(status),
    statusTone: statusToneFor(status),
    whyYouMatch: list.where((l) => l.tone == PresentationTone.positive).toList(),
    missingInformation: missingInformation,
    missingDocuments: missingDocuments,
    benefitSummary: benefitSummary,
    detailedInformation: details,
  );
}


class _StatusEntry {
  final String key;
  final String label;
  final PresentationTone tone;
  const _StatusEntry(this.key, this.label, this.tone);
}

Map<String, String> _labelForField(String field) {
  final k = field.trim().toLowerCase();
  return {
    'income_tax_payer': 'income tax status',
    'beneficiary_type': 'beneficiary type',
    'farmer_status': 'farmer status',
    'trade_license': 'trade licence',
    'age': 'age',
    'education': 'education',
    'income': 'income',
    'land_ownership': 'land ownership',
    'residence': 'residence',
  }[k] ?? k;
}

bool _likelyDocument(String field) {
  final t = field.toLowerCase();
  return t.contains('document') ||
      t.contains('aadhaar') ||
      t.contains('certificate') ||
      t.contains('ration') ||
      t.contains('land record');
}

ExplanationLine _buildLine(EligibilityRuleResult rule) {
  final raw = rule.displayText.trim();
  final fieldKey = rule.fieldName.trim();
  final friendly = _labelForField(fieldKey);

  if (rule.result == 'pass') {
    if (_likelyDocument(fieldKey)) {
      return ExplanationLine(
        'Your $friendly is on file and matches this scheme.',
        PresentationTone.positive,
      );
    }
    return ExplanationLine(
      'Your $friendly matches the scheme requirement.',
      PresentationTone.positive,
    );
  }

  if (rule.result == 'fail') {
    if (_likelyDocument(fieldKey)) {
      return ExplanationLine(
        'We still need your $friendly for this scheme.',
        PresentationTone.negative,
      );
    }
    return ExplanationLine(
      'Your $friendly does not match the scheme requirement.',
      PresentationTone.negative,
    );
  }

  if (rule.result == 'unknown') {
    return ExplanationLine(
      'We do not yet have your $friendly.',
      PresentationTone.neutral,
    );
  }

  if (rule.result == 'manual_review') {
    return ExplanationLine(
      'Your $friendly needs manual verification.',
      PresentationTone.caution,
    );
  }

  if (rule.result == 'insufficient') {
    return ExplanationLine(
      'We need more information about your $friendly.',
      PresentationTone.neutral,
    );
  }

  return ExplanationLine(
    'We need more information about your $friendly.',
    PresentationTone.neutral,
  );
}

DetailedRuleInfo _buildDetail(EligibilityRuleResult rule) {
  return DetailedRuleInfo(
    field: rule.fieldName,
    operatorName: rule.operator,
    expected: rule.expectedValue ?? '',
    actual: rule.actualValue ?? '',
    resultLabel: conditionResultLabel(rule.result),
    sourceDocument: rule.sourceDocument,
  );
}
